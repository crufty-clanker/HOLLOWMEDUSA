## 3.6 — Pipeline Runner Page

**Files:**

**Verification:**
```tsx
// frontend/src/pages/PipelineRunner.tsx
export function PipelineRunner() {
  const [run, setRun] = useRun()
  const wsRef = useRef<WebSocket | null>(null)

  useEffect(() => {
    wsRef.current = new WebSocket(`ws://${window.location.host}/runs/${run.id}/events`)
    wsRef.current.onmessage = (e) => {
      const event = JSON.parse(e.data)
      setRun(prev => ({ ...prev, step_results: event.step_results }))
    }
    return () => wsRef.current?.close()
  }, [run.id])

  return (
    <div className="space-y-4">
      <div className="flex gap-2">
        <Button onClick={startRun}>Start</Button>
        <Button onClick={pauseRun} disabled={!run?.running}>Pause</Button>
        <Button onClick={resumeRun} disabled={!run?.paused}>Resume</Button>
        <Button onClick={stopRun} disabled={!run?.running}>Stop</Button>
      </div>
      <StepProgress steps={run?.step_results || []} />
    </div>
  )
}
```
